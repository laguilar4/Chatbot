-- ---------------------------------------------------------------
-- 1. Crear base de datos
-- ---------------------------------------------------------------
CREATE DATABASE IF NOT EXISTS chatbot_db
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;

USE chatbot_db;

-- ---------------------------------------------------------------
-- 2. (Recomendado) Usuario dedicado con permisos mínimos
--    Cambia la contraseña antes de ejecutar.
-- ---------------------------------------------------------------
CREATE USER IF NOT EXISTS 'chatbot_user'@'localhost' IDENTIFIED BY 'CambiaEstaClave123!';
GRANT SELECT, INSERT, UPDATE, DELETE ON chatbot_db.* TO 'chatbot_user'@'localhost';
FLUSH PRIVILEGES;

-- ---------------------------------------------------------------
-- 3. Tabla de preguntas
--    option_number : número que el usuario teclea en el chat (único)
--    question      : texto de la pregunta mostrada en el menú
--    answer        : respuesta que muestra el bot
--    is_active     : permite ocultar una pregunta sin borrarla
-- ---------------------------------------------------------------
CREATE TABLE IF NOT EXISTS questions (
  id             INT UNSIGNED      NOT NULL AUTO_INCREMENT,
  option_number  SMALLINT UNSIGNED NOT NULL,
  question       VARCHAR(255)      NOT NULL,
  answer         TEXT              NOT NULL,
  is_active      TINYINT(1)        NOT NULL DEFAULT 1,
  created_at     TIMESTAMP         NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at     TIMESTAMP         NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uq_questions_option_number (option_number),
  KEY idx_questions_active_number (is_active, option_number),
  CONSTRAINT chk_questions_option_number CHECK (option_number > 0)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
